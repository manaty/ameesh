# Modules « pilote » : ceux que la CI d'une PR passe AUSSI avec le pilote psql
# (L116). Le reste de la suite ne dépend pas du pilote : il tourne avec psycopg
# sur la PR, et la suite psql complète tourne sur main, release/* et à la main
# (workflow_dispatch). Un module ajouté ici doit exister (scripts/test-parts.py
# refuse un nom inconnu).
#
# Règle : on liste les tests de la couche base (db.py), des migrations et du
# stockage (ameesh.storage), et ceux qui exercent un comportement propre à un
# pilote (connexion persistante, notifications, nombre d'allers-retours).
#
# Liste additive (L160a) : un module par ligne, sa raison en commentaire sur la
# même ligne, ordre alphabétique, sans doublon (scripts/test-parts.py refuse
# une liste mal rangée). Deux lots qui ajoutent chacun un module touchent ainsi
# deux endroits différents du fichier, sans conflit.

test_catalog_store  # stockage : le catalogue en base, opérations du domaine (paramètres en liste)
test_events_attach  # notifications : le listener psql persistant (le test force AMEESH_DRIVER=psql)
test_l122_numeros_de_migration  # stockage : réservation des numéros de migration (verrou consultatif, index unique) (L122)
test_l124_decisions  # stockage : domaine `decisions` (agent_mailbox, jsonb) et vue par projet (L124)
test_l130_demandes_sans_lot  # stockage : messages d'humains sans lot (agent_mailbox, jsonb, listes IN) (L130)
test_l157_estimations  # stockage : estimations (booléens liés, transactions, historique) (L157)
test_l157c_dates_mesures  # storage: phase measures (numeric, upsert with coalesce, LATERAL on milestones) and planned dates written (L157c)
test_l160d_resolutions  # stockage : résolutions préparées (text[] passé en jsonb, upsert, journal dans la même transaction) (L160d)
test_l162_tmpfs  # stockage : signalements de /tmp filtrés par `kind` (housekeeping.recent, L162)
test_l165_prise_lots  # stockage : lots à confier, équipe de la fiche, EXISTS sur décisions et jalons (booléens) (L165)
test_l368_planned_waits  # storage: unbounded activity, item plans and commitment dependencies (L368)
test_l165b_agents_libres  # stockage : `active_lots` du listing des agents, comptage et date prévue (L165b)
test_l317_project_modules  # storage: transactional module guards, structured evidence and project isolation (L317a)
test_l56_role_executor  # driver: the creator role is set for several statements, which needs one session (psql: transaction) (L56)
test_l61_allers_retours  # allers-retours comptés par le pilote lui-même (AMEESH_DB_TRACE)
test_l72_panne_de_base  # panne de base, reconnexion, réveils (listener)
test_migrations  # couche base : migrations sur un schéma neuf
test_registry_lease  # verrous et baux : comportement différent entre psql et psycopg
test_undeliverable  # courrier en souffrance : renvoi en une transaction (psql persistant), lecture jointe à la vue par projet (une requête)
